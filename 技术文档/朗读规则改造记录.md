**detectAvailableVoices 标签扩容兜底（改造点3）**
```javascript
// 配置项可能有序号超出范围的tag(如女青年517)，GENSHIN遍历不到
if (tagsData) {
    var _extPrefixes = {};
    for (var _ek in GENSHIN_CHARACTERS) {
        if (GENSHIN_CHARACTERS.hasOwnProperty(_ek)) {
            var _evp = GENSHIN_CHARACTERS[_ek].voice.toString().replace(/\d+$/, '');
            if (_evp) _extPrefixes[_evp] = true;
        }
    }
    for (var _tdKey in tagsData) {
        if (!Object.prototype.hasOwnProperty.call(tagsData, _tdKey)) continue;
        if (this.availableVoices[_tdKey]) continue;
        var _em = _tdKey.match(/^(.+?)(\d+)$/);
        if (_em && _extPrefixes[_em[1]]) {
            this.availableVoices[_tdKey] = true;
        }
    }
}
```

## 二、直连版改造（仅需改1）

需改1原来通过 HTTP 接口获取章节内容，与需改2/原版不一致。直连版改为读取本地 `data.json`，与需改2/原版对齐。

### 改造点

| 改造点 | 改造前（需改1原版） | 改造后（直连版） |
|--------|---------------------|-----------------|
| 书架获取 | `ttsrv.httpGet("http://localhost:1122/getBookshelf")` | `ttsrv.readTxtFile("data.json")` |
| 书名获取 | `firstBook = bookshelfData.data[0]` | `firstBook.name = bookData.bookName` |
| 章节内容 | 循环 `httpGet getBookContent` 加载当前+后续N章 | 直接 `bookData.texts` 字段 |
| bookUrl | 从书架接口获取 | 从data.json读取，缺失兜底为空字符串 |
| durChapterIndex | 从书架接口获取 | 从data.json读取，缺失兜底为0 |

### 关键代码片段

**书架获取替换**
```javascript
// 改造前
var bookshelfResponse = ttsrv.httpGet("http://localhost:1122/getBookshelf");
if (bookshelfResponse && bookshelfResponse.body()) {
    var bookshelfData = JSON.parse(bookshelfResponse.body().string());
    if (bookshelfData.isSuccess && bookshelfData.data && bookshelfData.data.length > 0) {
        var firstBook = bookshelfData.data[0];
        var newBookName = String(firstBook.name).trim();
        // ...
    }
}

// 改造后
var dataJsonContent = "";
try {
    dataJsonContent = ttsrv.readTxtFile("data.json").toString();
} catch (e) {}
if (dataJsonContent && dataJsonContent.trim() !== "") {
    var bookData = JSON.parse(dataJsonContent.toString());
    var firstBook = {
        name: String(bookData.bookName || "未知书名").trim(),
        bookUrl: bookData.bookUrl ? String(bookData.bookUrl) : "",
        durChapterIndex: (typeof bookData.durChapterIndex !== 'undefined') ? bookData.durChapterIndex : 0
    };
    var newBookName = firstBook.name;
    // ...
}
```

**章节内容获取替换**
```javascript
// 改造前：循环加载当前章节+后续NEXT_CHAPTER_COUNT章
var fullChapterContent = "";
for (var i = 0; i <= NEXT_CHAPTER_COUNT; i++) {
    var targetIndex = currentChapterIndex + i;
    var chapterUrl = "http://localhost:1122/getBookContent?url=" + bookUrl + "&index=" + targetIndex;
    var chapterResponse = ttsrv.httpGet(chapterUrl);
    if (chapterResponse && chapterResponse.body()) {
        var chapterData = JSON.parse(chapterResponse.body().string().toString());
        if (chapterData.isSuccess && chapterData.data) {
            fullChapterContent += String(chapterData.data).toString() + "\n";
        }
    }
}

// 改造后：直接读取data.json的texts字段
var fullChapterContent = String(bookData.texts || "").toString();
```

### 兼容处理

- `firstBook.bookUrl` 和 `firstBook.durChapterIndex` 保留（`graphSetCurrentChapterKey` 依赖），从 data.json 读取并兜底
- 换书逻辑、上下文匹配逻辑、文本处理逻辑全部保留不变
- 移除了 `NEXT_CHAPTER_COUNT` 多章节循环（data.json 的 texts 已是当前章节内容）

### data.json 字段说明

| 字段 | 类型 | 说明 |
|------|------|------|
| bookName | string | 书名（对应原firstBook.name） |
| texts | string | 章节正文内容（对应原getBookContent返回的data） |
| bookUrl | string | 书籍URL（可选，用于graphSetCurrentChapterKey） |
| durChapterIndex | number | 当前章节序号（可选，用于graphSetCurrentChapterKey） |
